      
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
OAICS_BA_Response_Extractor.py

离线版：从你已经保存下来的 Checkout / Stripe / OpenAI 响应或日志中，
识别 OAICS，并提取已经存在于响应里的 PayPal BA approve URL / ba_token。

不会：
- 发起任何网络请求
- 使用 AT 登录或创建 Checkout
- 调用 OpenAI checkout/create/update/confirm
- 调用 Stripe init/confirm
- 创建 SetupIntent / PaymentMethod
- 跟随重定向
- 执行 PayPal Agreement 或支付

适合用来验证：
1. 当前响应链是否属于 oaics_ Checkout
2. 响应中是否已经出现 PayPal agreements/approve
3. 是否能提取已有 ba_token
4. SetupIntent / PaymentIntent / redirect 信息是否已经存在
"""

from __future__ import annotations

import argparse
import json
import re
from pathlib import Path
from typing import Any
from urllib.parse import urlsplit, parse_qsl


OAICS_RE = re.compile(r"\boaics_[A-Za-z0-9_-]+\b")
CS_RE = re.compile(r"\bcs_(?:live|test)?_[A-Za-z0-9_-]+\b|\bcs_[A-Za-z0-9_-]+\b")
PAYPAL_BA_URL_RE = re.compile(
    r"https://(?:www\.)?paypal\.com/agreements/approve\?[^\s\"'<>]+",
    re.I,
)
BA_TOKEN_RE = re.compile(r"\bBA-[A-Za-z0-9_-]+\b", re.I)
SETUP_INTENT_RE = re.compile(r"\bseti_[A-Za-z0-9_-]+\b")
PAYMENT_INTENT_RE = re.compile(r"\bpi_[A-Za-z0-9_-]+\b")
PM_REDIRECT_RE = re.compile(
    r"https://pm-redirects\.stripe\.com/authorize/[^\s\"'<>]+",
    re.I,
)


def load_any(path: Path) -> Any:
    text = path.read_text(encoding="utf-8", errors="replace")
    try:
        return json.loads(text)
    except Exception:
        return text


def walk(value: Any):
    """递归遍历 JSON 中的所有值。"""
    if isinstance(value, dict):
        for k, v in value.items():
            yield ("key", str(k))
            yield ("value", v)
            yield from walk(v)
    elif isinstance(value, list):
        for item in value:
            yield ("value", item)
            yield from walk(item)


def all_strings(value: Any) -> list[str]:
    out: list[str] = []

    if isinstance(value, str):
        out.append(value)
    elif isinstance(value, (dict, list)):
        for kind, item in walk(value):
            if kind == "value" and isinstance(item, str):
                out.append(item)

    return out


def dedupe(items: list[str]) -> list[str]:
    seen = set()
    out = []
    for item in items:
        if item and item not in seen:
            seen.add(item)
            out.append(item)
    return out


def mask_secret(value: str, keep_start: int = 8, keep_end: int = 6) -> str:
    value = str(value or "")
    if len(value) <= keep_start + keep_end + 3:
        return value
    return f"{value[:keep_start]}***{value[-keep_end:]}"


def validate_paypal_ba_url(url: str) -> bool:
    try:
        p = urlsplit(url)
    except Exception:
        return False

    host = (p.netloc or "").lower()
    if not (host == "paypal.com" or host.endswith(".paypal.com")):
        return False

    if p.path.rstrip("/").lower() != "/agreements/approve":
        return False

    q = dict(parse_qsl(p.query, keep_blank_values=True))
    token = str(q.get("ba_token") or "").strip()
    return token.upper().startswith("BA-")


def extract_ba_token_from_url(url: str) -> str:
    try:
        p = urlsplit(url)
        q = dict(parse_qsl(p.query, keep_blank_values=True))
        token = str(q.get("ba_token") or "").strip()
        if token.upper().startswith("BA-"):
            return token
    except Exception:
        pass
    return ""


def analyze_payload(payload: Any, source_name: str) -> dict[str, Any]:
    strings = all_strings(payload)
    joined = "\n".join(strings)

    oaics_ids = dedupe(OAICS_RE.findall(joined))
    cs_ids = dedupe(CS_RE.findall(joined))
    ba_urls = dedupe(PAYPAL_BA_URL_RE.findall(joined))
    valid_ba_urls = [u for u in ba_urls if validate_paypal_ba_url(u)]

    ba_tokens = []
    for u in valid_ba_urls:
        t = extract_ba_token_from_url(u)
        if t:
            ba_tokens.append(t)

    # 如果 token 单独出现在 JSON/log 中，也记录
    ba_tokens.extend(BA_TOKEN_RE.findall(joined))
    ba_tokens = dedupe(ba_tokens)

    setup_intents = dedupe(SETUP_INTENT_RE.findall(joined))
    payment_intents = dedupe(PAYMENT_INTENT_RE.findall(joined))
    pm_redirects = dedupe(PM_REDIRECT_RE.findall(joined))

    # 常见 next_action / redirect 字段，仅做离线读取
    redirect_candidates: list[str] = []
    if isinstance(payload, (dict, list)):
        for kind, item in walk(payload):
            if kind == "value" and isinstance(item, str):
                low = item.lower()
                if (
                    low.startswith("https://")
                    and ("paypal.com/" in low or "pm-redirects.stripe.com/" in low)
                ):
                    redirect_candidates.append(item)

    redirect_candidates = dedupe(redirect_candidates)

    return {
        "source": source_name,
        "oaics_present": bool(oaics_ids),
        "oaics_ids_masked": [mask_secret(x) for x in oaics_ids],
        "cs_ids_masked": [mask_secret(x) for x in cs_ids],
        "setup_intents_masked": [mask_secret(x) for x in setup_intents],
        "payment_intents_masked": [mask_secret(x) for x in payment_intents],
        "stripe_redirects_present": bool(pm_redirects),
        "paypal_ba_present": bool(valid_ba_urls or ba_tokens),
        "paypal_ba_urls": valid_ba_urls,
        "ba_tokens_masked": [mask_secret(x, 5, 5) for x in ba_tokens],
        "redirect_candidates": redirect_candidates,
    }


def merge_results(results: list[dict[str, Any]]) -> dict[str, Any]:
    oaics_present = any(r["oaics_present"] for r in results)
    ba_present = any(r["paypal_ba_present"] for r in results)

    if not oaics_present:
        state = "NOT_OAICS"
    elif ba_present:
        state = "OAICS_BA_FOUND"
    else:
        state = "OAICS_PRESENT_BA_NOT_FOUND"

    return {
        "state": state,
        "oaics_present": oaics_present,
        "paypal_ba_present": ba_present,
        "files_checked": len(results),
        "results": results,
    }


def collect_input_files(paths: list[str], folder: str | None) -> list[Path]:
    files: list[Path] = []

    for p in paths:
        path = Path(p)
        if path.is_file():
            files.append(path)

    if folder:
        root = Path(folder)
        if root.is_dir():
            for ext in ("*.json", "*.txt", "*.log"):
                files.extend(sorted(root.glob(ext)))

    # 去重
    seen = set()
    out = []
    for f in files:
        rf = f.resolve()
        if rf not in seen:
            seen.add(rf)
            out.append(f)

    return out


def main():
    parser = argparse.ArgumentParser(
        description="Offline OAICS -> existing PayPal BA response extractor"
    )
    parser.add_argument(
        "files",
        nargs="*",
        help="已有响应文件，可传多个：checkout.json stripe.json confirm.json log.txt",
    )
    parser.add_argument(
        "--folder",
        help="批量扫描目录中的 .json/.txt/.log 文件",
    )
    parser.add_argument(
        "--output",
        help="可选：把结果保存为 JSON",
    )
    args = parser.parse_args()

    files = collect_input_files(args.files, args.folder)

    if not files:
        raise SystemExit("没有找到输入文件。")

    results = []
    for path in files:
        payload = load_any(path)
        results.append(analyze_payload(payload, path.name))

    merged = merge_results(results)
    text = json.dumps(merged, ensure_ascii=False, indent=2)
    print(text)

    if args.output:
        Path(args.output).write_text(text, encoding="utf-8")


if __name__ == "__main__":
    main()
